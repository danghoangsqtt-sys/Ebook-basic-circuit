2. State Machine — Quản lý trạng thái

// state_machine.h — Ví dụ cho Trạm Thời Tiết IoT
enum SystemState {
  STATE_INIT,
  STATE_CONNECTING,
  STATE_IDLE,
  STATE_READING_SENSORS,
  STATE_UPLOADING,
  STATE_ALARM,
  STATE_ERROR
};

// main.cpp — State machine loop
SystemState currentState = STATE_INIT;
unsigned long lastSensorRead = 0;
unsigned long lastUpload = 0;

void loop() {
  switch (currentState) {
    case STATE_INIT:
      initAllModules();
      currentState = STATE_CONNECTING;
      break;
      
    case STATE_CONNECTING:
      if (connectWiFi(5000)) currentState = STATE_IDLE;
      else                   currentState = STATE_ERROR;
      break;
      
    case STATE_IDLE:
      if (millis() - lastSensorRead > 5000) {
        currentState = STATE_READING_SENSORS;
      }
      break;
      
    case STATE_READING_SENSORS:
      readAllSensors();  // Đọc BME280, v.v.
      lastSensorRead = millis();
      display.update();
      if (checkAlarmConditions()) currentState = STATE_ALARM;
      else if (millis() - lastUpload > 60000) currentState = STATE_UPLOADING;
      else currentState = STATE_IDLE;
      break;
      
    case STATE_UPLOADING:
      uploadToAdafruitIO();
      lastUpload = millis();
      currentState = STATE_IDLE;
      break;
      
    case STATE_ALARM:
      sendTelegramAlert();
      currentState = STATE_IDLE;
      break;
      
    case STATE_ERROR:
      Serial.println("[ERROR] System error. Retrying...");
      delay(5000);
      ESP.restart();  // Hard reset sau lỗi
      break;
  }
}

3. Task Scheduler đơn giản — millis() pattern

// Non-blocking task scheduler — không dùng delay()
struct Task {
  unsigned long interval;
  unsigned long lastRun;
  void (*handler)();
};

Task tasks[] = {
  {2000,  0, readSensors},   // Mỗi 2s
  {1000,  0, updateDisplay}, // Mỗi 1s
  {60000, 0, uploadCloud},   // Mỗi 60s
  {10000, 0, logToSD},       // Mỗi 10s
  {500,   0, checkButtons},  // Mỗi 500ms
};

void loop() {
  unsigned long now = millis();
  for (auto& t : tasks) {
    if (now - t.lastRun >= t.interval) {
      t.lastRun = now;
      t.handler();
    }
  }
}

Tại sao pattern millis()-based scheduler tốt hơn delay() trong hệ thống nhúng?

3: delay() chặn toàn bộ CPU trong khoảng thời gian đó — không thể làm gì khác (không đọc sensor khác, không check WiFi, không xử lý interrupt từ nút nhấn). millis()-based scheduler: mỗi task chỉ chạy khi đến lượt, CPU luôn tự do giữa các task. Kết quả: hệ thống phản hồi (responsive) và có thể xử lý nhiều task đồng thời giả lập (cooperative multitasking).
