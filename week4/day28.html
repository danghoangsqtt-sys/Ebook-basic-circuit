2. Mini Project Tuần 4: Trạm Đo Môi Trường

📋 Yêu cầu kỹ thuật
  • Cảm biến: DHT22 (nhiệt độ + độ ẩm) hoặc BME280 I2C
  • Hiển thị: OLED SSD1306 0,96" I2C — nhiệt độ, độ ẩm, thời gian chạy
  • Log dữ liệu: SD card SPI — ghi mỗi 10 giây: timestamp,temp,humi
  • Cảnh báo: LED đỏ + buzzer khi nhiệt độ > 35°C
  • Nút nhấn: 1 nút (interrupt) để export summary qua Serial
  • Nguồn: USB 5V → AMS1117-3.3 → toàn bộ mạch

2.1 Kiến trúc phần mềm

Sơ đồ task / luồng xử lý: setup(): ├─ Init Serial (115200) ├─ Init I2C → OLED.begin() → BME280.begin() ├─ Init SPI → SD.begin() ├─ attachInterrupt(btnPin, onBtn, FALLING) └─ Init timer hoặc millis-based scheduler loop(): ├─ Mỗi 2000ms: đọc DHT22/BME280 │ ├─ Validate: NaN check │ ├─ Update OLED display │ └─ Nếu temp > 35: LED ON + buzzer beep ├─ Mỗi 10000ms: ghi SD card │ └─ Format: "millis,temp,humi\n" └─ Nếu btnFlag: in summary qua Serial ISR onBtn(): └─ btnFlag = true (chỉ set flag, không xử lý)

2.2 Code khung (ESP32)

// Trạm Đo Môi Trường — ESP32 Skeleton
#include <Wire.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>
#include <SPI.h>
#include <SD.h>

#define DHT_PIN  4   #define BTN_PIN  0
#define LED_PIN  2   #define CS_SD    5
#define BUZZER   15

DHT dht(DHT_PIN, DHT22);
Adafruit_SSD1306 oled(128, 64, &Wire, -1);

volatile bool btnFlag = false;
unsigned long lastSensor = 0, lastLog = 0;
float temp = 0, humi = 0;

void IRAM_ATTR onBtn() { btnFlag = true; }

void setup() {
  Serial.begin(115200);
  dht.begin();
  oled.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  oled.clearDisplay();
  SD.begin(CS_SD);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  attachInterrupt(BTN_PIN, onBtn, FALLING);
}

void loop() {
  unsigned long now = millis();
  
  // Đọc cảm biến mỗi 2s
  if (now - lastSensor >= 2000) {
    lastSensor = now;
    float t = dht.readTemperature();
    float h = dht.readHumidity();
    if (!isnan(t) && !isnan(h)) { temp = t; humi = h; }
    
    // Update OLED
    oled.clearDisplay();
    oled.setTextSize(2); oled.setCursor(0,0);
    oled.printf("%.1fC", temp);
    oled.setTextSize(1); oled.setCursor(0,35);
    oled.printf("Humi: %.1f%%", humi);
    oled.display();
    
    // Cảnh báo nhiệt
    bool hot = (temp > 35.0);
    digitalWrite(LED_PIN, hot);
    if (hot) { tone(BUZZER, 1000, 200); }
  }
  
  // Log SD mỗi 10s
  if (now - lastLog >= 10000) {
    lastLog = now;
    File f = SD.open("/log.csv", FILE_APPEND);
    if (f) { f.printf("%lu,%.2f,%.2f\n", now, temp, humi); f.close(); }
  }
  
  // Nút nhấn summary
  if (btnFlag) {
    btnFlag = false;
    Serial.printf("SUMMARY — Temp: %.1fC  Humi: %.1f%%\n", temp, humi);
  }
}
